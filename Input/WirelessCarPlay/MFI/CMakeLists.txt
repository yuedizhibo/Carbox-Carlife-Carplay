add_library(auth3_native_core auth3_core.cpp auth3_safety.cpp)
target_include_directories(auth3_native_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(auth3_native_core PUBLIC cxx_std_20)
if(MSVC)
  target_compile_options(auth3_native_core PRIVATE /W4 /permissive-)
else()
  target_compile_options(auth3_native_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
endif()

add_executable(auth3_native_tests auth3_tests.cpp)
target_link_libraries(auth3_native_tests PRIVATE auth3_native_core)
if(MSVC)
  target_compile_options(auth3_native_tests PRIVATE /W4 /permissive-)
else()
  target_compile_options(auth3_native_tests PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
endif()
add_test(NAME auth3_native_tests COMMAND auth3_native_tests)

# The hardware tool intentionally exists only on Linux; tests remain portable.
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  add_executable(auth3-native auth3_native.cpp)
  target_link_libraries(auth3-native PRIVATE auth3_native_core)
  if(MSVC)
    target_compile_options(auth3-native PRIVATE /W4 /permissive-)
  else()
    target_compile_options(auth3-native PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
  endif()
endif()
