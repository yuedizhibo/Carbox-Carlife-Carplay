cmake_minimum_required(VERSION 3.20)
project(wired_carplay_output LANGUAGES CXX)

add_library(wired_carplay_output src/output.cpp)
target_compile_features(wired_carplay_output PUBLIC cxx_std_20)
target_include_directories(wired_carplay_output PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
find_package(Threads REQUIRED)
target_link_libraries(wired_carplay_output PUBLIC Threads::Threads)
if(MSVC)
  target_compile_options(wired_carplay_output PRIVATE /W4)
else()
  target_compile_options(wired_carplay_output PRIVATE -Wall -Wextra -Wpedantic)
endif()
option(WIRED_CARPLAY_BUILD_TESTS "Build standalone Output contract tests" ON)
if(WIRED_CARPLAY_BUILD_TESTS)
  enable_testing()
  add_executable(wired_output_tests tests/output_tests.cpp)
  target_link_libraries(wired_output_tests PRIVATE wired_carplay_output)
  add_test(NAME wired_output_tests COMMAND wired_output_tests)
  add_executable(wired_catalog_tests tests/catalog_tests.cpp)
  target_link_libraries(wired_catalog_tests PRIVATE wired_carplay_output)
  add_test(NAME wired_catalog_tests COMMAND wired_catalog_tests)
  set_tests_properties(wired_output_tests wired_catalog_tests PROPERTIES TIMEOUT 20)
endif()
